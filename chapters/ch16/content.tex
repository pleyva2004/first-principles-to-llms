\subsection*{Motivation}

A multi-layer perceptron (Chapter~15) without a nonlinear activation collapses into a single affine map: $W_2(W_1 x + b_1) + b_2 = (W_2 W_1) x + (W_2 b_1 + b_2)$. Universal approximation requires a pointwise nonlinearity $\phi$ between affine layers. The choice of $\phi$ controls the gradient signal that backpropagation (Chapter~3) is allowed to push through the network, the geometry of hidden activations, and the numerical conditioning of training. We derive the five activations that dominate modern deep learning---sigmoid, tanh, ReLU, GELU, and softmax---computing each derivative or Jacobian from first principles and analyzing saturation.

\subsection*{Definitions}

\begin{definition}[Sigmoid]
$\sigma:\mathbb{R}\to(0,1)$, $\sigma(x) = \dfrac{1}{1+e^{-x}}$.
\end{definition}

\begin{definition}[Hyperbolic tangent]
$\tanh(x) = \dfrac{e^x-e^{-x}}{e^x+e^{-x}} = 2\sigma(2x)-1$.
\end{definition}

\begin{definition}[ReLU]
$\mathrm{ReLU}(x) = \max(0,x)$. Differentiable except at $x=0$; the Clarke subdifferential at $0$ is $[0,1]$.
\end{definition}

\begin{definition}[GELU]
$\mathrm{GELU}(x) = x\,\Phi(x)$, where $\Phi(x) = \tfrac{1}{2}\bigl(1 + \mathrm{erf}(x/\sqrt{2})\bigr)$ is the standard-normal CDF. The tanh-approximation is
\[
\mathrm{GELU}(x) \;\approx\; \tfrac{1}{2}x\bigl(1 + \tanh(\sqrt{2/\pi}\,(x + 0.044715\,x^3))\bigr).
\]
\end{definition}

\begin{definition}[Softmax]
$\mathrm{softmax}:\mathbb{R}^n\to\Delta^{n-1}$, $\mathrm{softmax}(z)_i = \dfrac{e^{z_i}}{\sum_{k=1}^n e^{z_k}}$.
\end{definition}

\subsection*{Theorems and Proofs}

\begin{theorem}[Sigmoid derivative]\label{thm:sig}
$\sigma'(x) = \sigma(x)\bigl(1-\sigma(x)\bigr)$.
\end{theorem}
\begin{proof}
Write $\sigma(x)=(1+e^{-x})^{-1}$. By the chain rule,
$\sigma'(x) = -(1+e^{-x})^{-2}\cdot(-e^{-x}) = \dfrac{e^{-x}}{(1+e^{-x})^2}$. Factor:
\[
\frac{e^{-x}}{(1+e^{-x})^2} = \frac{1}{1+e^{-x}}\cdot\frac{e^{-x}}{1+e^{-x}} = \sigma(x)\bigl(1-\sigma(x)\bigr),
\]
using $\frac{e^{-x}}{1+e^{-x}} = 1-\sigma(x)$.
\end{proof}

\begin{theorem}[Tanh derivative]
$\tanh'(x) = 1-\tanh^2(x)$.
\end{theorem}
\begin{proof}
From $\tanh(x) = 2\sigma(2x)-1$ and Theorem~\ref{thm:sig},
$\tanh'(x) = 4\sigma(2x)(1-\sigma(2x))$. Substituting $\sigma(2x) = (\tanh x + 1)/2$,
\[
4\cdot\tfrac{\tanh x+1}{2}\cdot\tfrac{1-\tanh x}{2} = (1+\tanh x)(1-\tanh x) = 1-\tanh^2 x. \qedhere
\]
\end{proof}

\begin{theorem}[ReLU subdifferential]
For $x\neq 0$, $\mathrm{ReLU}'(x) = \mathbf{1}_{x>0}$. At $x=0$, $\partial\,\mathrm{ReLU}(0) = [0,1]$.
\end{theorem}
\begin{proof}
For $x>0$, $\mathrm{ReLU}(x)=x$ has derivative $1$. For $x<0$, derivative $0$. At $x=0$ the left derivative is $0$ and the right derivative is $1$; the convex hull of these one-sided limits, $[0,1]$, defines the Clarke subdifferential. PyTorch and JAX conventionally pick $0$.
\end{proof}

\begin{theorem}[GELU derivative]
$\mathrm{GELU}'(x) = \Phi(x) + x\,\phi(x)$, where $\phi(x) = \tfrac{1}{\sqrt{2\pi}}e^{-x^2/2}$.
\end{theorem}
\begin{proof}
By the product rule, $\frac{d}{dx}\bigl[x\Phi(x)\bigr] = \Phi(x) + x\,\Phi'(x)$. The fundamental theorem of calculus gives $\Phi'(x) = \phi(x)$.
\end{proof}

\begin{theorem}[Softmax Jacobian]\label{thm:smj}
Let $s = \mathrm{softmax}(z)$ and $S = \sum_k e^{z_k}$. Then
\[
\frac{\partial s_i}{\partial z_j} = s_i(\delta_{ij} - s_j), \qquad J(z) = \mathrm{diag}(s) - s s^\top.
\]
\end{theorem}
\begin{proof}
By the quotient rule,
\[
\frac{\partial s_i}{\partial z_j} = \frac{(\partial e^{z_i}/\partial z_j)\,S - e^{z_i}\,(\partial S/\partial z_j)}{S^2} = \frac{\delta_{ij}\,e^{z_i} S - e^{z_i} e^{z_j}}{S^2} = \delta_{ij}\,s_i - s_i s_j. \qedhere
\]
\end{proof}

The matrix $J = \mathrm{diag}(s) - ss^\top$ is symmetric PSD with $J\mathbf{1} = 0$, reflecting softmax's translation invariance.

\begin{proposition}[Saturation]
$\sigma'(x), \tanh'(x) \to 0$ as $|x|\to\infty$, so backpropagated gradients $\prod_\ell \phi'(z_\ell)$ shrink geometrically in deep networks (vanishing gradients). ReLU avoids saturation for $x>0$ but a unit with $z\le 0$ on every training input is permanently dead. GELU is smooth; $\mathrm{GELU}'(x)\to 1$ as $x\to+\infty$ (since $\Phi\to 1$, $x\phi\to 0$) and $\to 0$ as $x\to-\infty$, combining ReLU-like sparsity with smoothness.
\end{proposition}

\begin{theorem}[Temperature limits]
Let $T>0$ and assume $i^* = \arg\max_i z_i$ is unique. Then
\[
\lim_{T\to 0^+} \mathrm{softmax}(z/T)_i = \mathbf{1}_{i=i^*}, \qquad \lim_{T\to\infty}\mathrm{softmax}(z/T)_i = \tfrac{1}{n}.
\]
\end{theorem}
\begin{proof}
Write $\mathrm{softmax}(z/T)_i = \dfrac{e^{(z_i - z_{i^*})/T}}{\sum_k e^{(z_k - z_{i^*})/T}}$. As $T\to 0^+$, every term with $k\ne i^*$ has exponent $\to-\infty$, so $\to 0$, while $k=i^*$ contributes $1$, yielding $\mathbf{1}_{i=i^*}$. As $T\to\infty$, every exponent $\to 0$, so each term $\to 1$ and the ratio $\to 1/n$.
\end{proof}

\begin{proposition}[Numerical stability]
$\mathrm{softmax}(z) = \mathrm{softmax}(z - c\mathbf{1})$ for any $c$. With $c = \max_i z_i$, all exponents are $\le 0$ and overflow is avoided.
\end{proposition}
\begin{proof}
$\dfrac{e^{z_i - c}}{\sum_k e^{z_k - c}} = \dfrac{e^{-c}e^{z_i}}{e^{-c}\sum_k e^{z_k}} = \mathrm{softmax}(z)_i$.
\end{proof}

\begin{corollary}[Log-sum-exp]
$\log\sum_j e^{z_j} = \max_j z_j + \log\sum_j e^{z_j - \max_k z_k}$.
\end{corollary}

\subsection*{Connection to LLMs}

GELU is the standard feedforward activation in GPT-2/3 and BERT, applied elementwise to the $4d_{\text{model}}$ hidden state of every Transformer MLP block (Chapter~15). Llama-family models replace it with SwiGLU, a gated variant covered in Chapter~17. Softmax appears in two distinct loci of an LLM:

\begin{enumerate}
\item \emph{Attention} (Chapter~21): row-wise $\mathrm{softmax}(qK^\top/\sqrt{d_k})$ converts compatibility scores into a distribution over keys.
\item \emph{Output head} (Chapter~25): final vocabulary logits are passed through softmax to produce $p(x_{t+1}\mid x_{\le t})$. The temperature limits (Theorem above) are the standard sampling knob: $T\to 0$ recovers greedy decoding, $T\to\infty$ uniform random sampling.
\end{enumerate}

Both invocations use the stabilized form of Proposition above in production kernels (FlashAttention, fused log-softmax + cross-entropy), making the ``subtract the max'' identity load-bearing for trillion-parameter training.
